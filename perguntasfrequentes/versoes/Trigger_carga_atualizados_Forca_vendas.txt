--Trigger para que seja listado o produto na carga de atualizados

CREATE OR ALTER TRIGGER TG_PRODUTO_INTEGRACAOEXP FOR TPRODUTO
ACTIVE AFTER INSERT OR UPDATE POSITION 0
AS
  DECLARE VARIABLE QTD INTEGER;
  DECLARE VARIABLE CODPDV VARCHAR(10);
BEGIN
  IF (((NEW.CUSTOUNITARIO <> OLD.CUSTOUNITARIO) OR
    (NEW.NOMEFANTASIA <> OLD.NOMEFANTASIA) OR
    (NEW.NUMNOFABRIC <> OLD.NUMNOFABRIC) OR
    (NEW.CODBARRAS <> OLD.CODBARRAS) OR
    (NEW.CEST <> OLD.CEST) OR
    (NEW.UNIDADE <> OLD.UNIDADE) OR
    (NEW.CODUNDCOMPRA <> OLD.CODUNDCOMPRA) OR
    (NEW.CODUNDVENDA <> OLD.CODUNDVENDA) OR
    (NEW.PRECO1 <> OLD.PRECO1) OR
    (NEW.IDPROMOCAO IS NULL AND OLD.IDPROMOCAO IS NOT NULL) OR
    (NEW.IDPROMOCAO IS NOT NULL AND OLD.IDPROMOCAO IS NULL) OR
    (NEW.IDPROMOCAO <> OLD.IDPROMOCAO) OR
    (NEW.NAOEXPORTARPALM IS NULL AND OLD.NAOEXPORTARPALM IS NOT NULL) OR
    (NEW.NAOEXPORTARPALM IS NOT NULL AND OLD.NAOEXPORTARPALM IS NULL) OR
    (NEW.NAOEXPORTARPALM <> OLD.NAOEXPORTARPALM) OR
    (NEW.USAINTEGRACAO <> OLD.USAINTEGRACAO) OR
    (NEW.CODSIT <> OLD.CODSIT) OR
    (NEW.CSTPIS <> OLD.CSTPIS) OR
    (NEW.CODCLAS <> OLD.CODCLAS) OR
    (NEW.INATIVO <> OLD.INATIVO) OR
    (NEW.ORIGEMMERCADORIA <> OLD.ORIGEMMERCADORIA) OR
    (NEW.EXPORTABALANCA <> OLD.EXPORTABALANCA) OR
    (NEW.USAREGRAPRECO <> COALESCE(OLD.USAREGRAPRECO, '')) OR
    (NEW.NUMERONOCFO <> OLD.NUMERONOCFO) OR
    ((NEW.SALDOGERALFISICO > 0) and (OLD.SALDOGERALFISICO <= 0)) or
    ((NEW.SALDOGERALFISICO <= 0) and (OLD.SALDOGERALFISICO > 0))) or
    (INSERTING)) THEN
  BEGIN
    FOR SELECT T.CODPONTOVENDA FROM TPONTOVENDA T WHERE T.CODEMPRESA = NEW.CODEMPRESA AND
      COALESCE(T.INATIVO, '') <> 'T'
    INTO CODPDV DO
    BEGIN
      SELECT COUNT(D.CHAVE) FROM TINTEGRACAOEXP D WHERE D.SYNC_FLAG = '1' AND D.CHAVE = NEW.CODPRD AND D.ORIGEM = 'PRODUTO'
        AND D.CODPONTOVENDA = :CODPDV
      INTO QTD;
      IF (QTD = 0) THEN
        UPDATE OR INSERT INTO TINTEGRACAOEXP (ORIGEM, CODEMPRESA, CHAVE, SYNC_FLAG, SYNC_FLAG_FORCADEVENDA, CODPONTOVENDA)
          VALUES ('PRODUTO', NEW.CODEMPRESA, NEW.CODPRD, '1', 1, :CODPDV)
          MATCHING (CHAVE, CODEMPRESA, CODPONTOVENDA);
    END

    SELECT COUNT(D.CHAVE) FROM TINTEGRACAOEXP D WHERE D.SYNC_FLAG_FORCADEVENDA = 1 AND D.CHAVE = NEW.CODPRD AND D.ORIGEM = 'PRODUTO'
    INTO QTD;
    IF (QTD = 0) THEN
      UPDATE OR INSERT INTO TINTEGRACAOEXP (ORIGEM, CODEMPRESA, CHAVE, SYNC_FLAG,  SYNC_FLAG_FORCADEVENDA, CODPONTOVENDA)
        VALUES ('PRODUTO', NEW.CODEMPRESA, NEW.CODPRD, '1', 1, '' )
        MATCHING (CHAVE, CODEMPRESA, CODPONTOVENDA);
  END
END;


--Trigger para que seja listado o produto na carga de atualizados de imagens 
CREATE OR ALTER TRIGGER TG_PROD_IMG_INTEGRACAOEXP FOR TPRODIMAGEM
ACTIVE AFTER INSERT OR UPDATE OR DELETE POSITION 0
AS
  DECLARE VARIABLE QTD INTEGER;
  DECLARE VARIABLE CODPDV VARCHAR(10);
BEGIN
IF ((NEW.IMAGEM <> OLD.IMAGEM) OR
    (NEW.ORDEMPESQUISA IS NULL AND OLD.ORDEMPESQUISA IS NOT NULL) OR
    (NEW.ORDEMPESQUISA IS NOT NULL AND OLD.ORDEMPESQUISA IS NULL) OR
    (INSERTING)) THEN
  BEGIN
    SELECT COUNT(D.CHAVE) FROM TINTEGRACAOEXP D WHERE D.SYNC_FLAG_FORCADEVENDA = 1 AND D.CHAVE = NEW.CODPRD AND D.ORIGEM = 'PRODIMAGEM'
    INTO QTD;
    IF (QTD = 0) THEN
      UPDATE OR INSERT INTO TINTEGRACAOEXP (ORIGEM, CODEMPRESA, CHAVE, SYNC_FLAG,  SYNC_FLAG_FORCADEVENDA, CODPONTOVENDA)
        VALUES ('PRODIMAGEM', NEW.CODEMPRESA, NEW.CODPRD, '1', 1, '' )
        MATCHING (CHAVE, CODEMPRESA, CODPONTOVENDA);
  END
  IF (DELETING) THEN
  UPDATE OR INSERT INTO TINTEGRACAOEXP (ORIGEM, CODEMPRESA, CHAVE, SYNC_FLAG,  SYNC_FLAG_FORCADEVENDA, CODPONTOVENDA)
        VALUES ('PRODIMAGEM', OLD.CODEMPRESA, OLD.CODPRD, '1', 1, '' )
        MATCHING (CHAVE, CODEMPRESA, CODPONTOVENDA);
END;
