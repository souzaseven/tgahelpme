        function hasWhatsAppCache() {
            var whatsappCache = localStorage.getItem("hasWhatsapp");

            if (!whatsappCache || whatsappCache.expiry < new Date().getTime())
                return false;

            return true;
        }

        function setWhatsAppCache(hasWhatsApp) {
            localStorage.setItem("hasWhatsapp", JSON.stringify({
                hasWhatsapp: hasWhatsApp,
                expiry: new Date().getTime() + 80000000
            }));
        }

        function pushToDataLayer(hasWhatsapp) {
            window.dataLayer = window.dataLayer || [];
            window.dataLayer.push({
                'Whatsapp' : hasWhatsapp
            });
        }
